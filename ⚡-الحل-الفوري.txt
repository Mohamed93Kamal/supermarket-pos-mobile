import os

# إنشاء buildozer.spec تلقائياً
if not os.path.exists('buildozer.spec'):
    print("📝 إنشاء buildozer.spec...\n")
    buildozer_content = """[app]
title = نظام كاشير سوبر ماركت
package.name = supermarketpos
package.domain = org.supermarket

source.dir = .
source.include_exts = py,png,jpg,kv,atlas,db
source.main = app_final.py

version = 3.0.0

requirements = python3,kivy,kivymd

orientation = portrait
fullscreen = 0

android.permissions = INTERNET,WRITE_EXTERNAL_STORAGE,READ_EXTERNAL_STORAGE,CAMERA
android.api = 31
android.minapi = 21
android.ndk = 25b
android.accept_sdk_license = True

[buildozer]
log_level = 2
warn_on_root = 1
"""
    with open('buildozer.spec', 'w', encoding='utf-8') as f:
        f.write(buildozer_content)
    print("✅ تم إنشاء buildozer.spec!\n")

# التحقق
required_files = ['app_final.py', 'buildozer.spec', 'supermarket.db']

missing_files = [f for f in required_files if not os.path.exists(f)]

if missing_files:
    print("\n❌ الملفات التالية مفقودة:")
    for f in missing_files:
        print(f"   - {f}")
else:
    print("\n✅ جميع الملفات موجودة!")

from google.colab import files
import os

if not os.path.exists('bin'):
    print("❌ مجلد bin غير موجود - البناء فشل!")
else:
    apk_files = [f for f in os.listdir('bin') if f.endswith('.apk')]
    if apk_files:
        files.download(f'bin/{apk_files[0]}')
        print(f"\n✅ تم تحميل: {apk_files[0]}")
        print("\n🎉 ثبّته على الموبايل!")
        print("📱 سجّل دخول: admin / admin123")
    else:
        print("❌ لم يتم العثور على APK!")
